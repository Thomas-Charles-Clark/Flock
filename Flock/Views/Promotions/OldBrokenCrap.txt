        function ChangeCellStyleOverride(visibleIndex, columnIndex, styleIndex) {
            if (window.cellEditError) {
                if (typeof (window.runCounter) == "undefined" || typeof (window.runCounter) == false)
                    window.runCounter = 0;
            }
            else if (window.runCounter > 2)
                window.runCounter = 0;

            if (styleIndex < 0 || styleIndex > 2)
                return;
            var row = this.grid.GetDataRow(visibleIndex);
            if (!row) return;
            var cell = this.GetDataCellByColumnIndex(row, columnIndex);
            if (!cell) return;
            if (styleIndex === 2) {
                if (_aspxIsExists(cell.dxgvSavedClassName))
                    cell.className = cell.dxgvSavedClassName;
                if (_aspxIsExists(cell.dxgvSavedCssText))
                    cell.style.cssText = cell.dxgvSavedCssText;
                return;
            }
            if (!_aspxIsExists(cell.dxgvSavedClassName))
                cell.dxgvSavedClassName = cell.className;
            if (!_aspxIsExists(cell.dxgvSavedCssText))
                cell.dxgvSavedCssText = cell.style.cssText;
            var styleRow = this.grid._getStyleRow(-1, 3);
            var styleCell = styleIndex === 0 ? styleRow.cells[0] : styleRow.cells[1];
            cell.className = styleCell.className;
            cell.style.cssText = styleCell.style.cssText;

            if (window.cellEditError) {
                cell.style.cssText += 'background-color: #ff0000';
                window.runCounter++;
            }
            if (window.runCounter > 0) {
                cell.style.cssText += 'background-color: #ff0000';
                $(cell).attr("title", window.errorTipText);
                window.runCounter++;
                window.cellEditError = false;
            }
            this.LoadCellAlign(cell, columnIndex);
        }


        function DateSanityCheck(OnDate, OffDate) {
            return OnDate < OffDate;
        }

        function date_by_subtracting_days(date, days) {
            return new Date(
                date.getFullYear(),
                date.getMonth(),
                date.getDate() - days,
                date.getHours(),
                date.getMinutes(),
                date.getSeconds(),
                date.getMilliseconds()
            );
        }

        function CalculateStatus(onDate, offDate) {
            var StatusID = -1;
            var PendingOn = new Date();
            var PendingOff = new Date();
            var Now = new Date();
            var Offset = 14;
            var StatusText = "Unknown";

            PendingOn = date_by_subtracting_days(onDate, Offset - onDate.getDate());//onDate.addDays(-Offset);

            if (offDate != null) {
                PendingOff = date_by_subtracting_days(offDate, Offset - offDate.getDate());

                if (Now < PendingOn)
                    StatusID = 3;
                else if (Now > PendingOn && Now < onDate)
                    StatusID = 2;
                else if (Now > PendingOff && Now < offDate)
                    StatusID = 1;
                else if (Now > onDate && Now < PendingOff)
                    StatusID = 0;
                else if (Now >= onDate && offDate == null)
                    StatusID = 0;
                else
                    StatusID = 4;
            }
            else {
                if (Now < PendingOn)
                    StatusID = 3;
                else if (Now > PendingOn && Now < onDate)
                    StatusID = 2;
                else if (Now >= onDate && offDate == null)
                    StatusID = 0;
                else
                    StatusID = 4;
            }

            switch (StatusID) {
                case 0:
                    StatusText = "On";
                    break;
                case 1:
                    StatusText = "Pending Off (On)";
                    break;
                case 2:
                    StatusText = "Pending On";
                    break;
                case 3:
                    StatusText = "Pre-Launch";
                    break;
                case 4:
                    StatusText = "Off";
                    break;
            }

            var result = [];
            result.StatusID = StatusID;
            result.StatusText = StatusText;

            return result;
        }

            function BulkUpdateValuesClientSide(key, row, shift) {

                var ExRate = $("#price-list-exchange-rate").html();
                var beh = SalePriceAddNewGridView.GetBatchEditHelper();
                beh.updatedValues[key] = new Object();
                beh.updatedValues[key][(19 + shift).toString()] = new Array();
                beh.updatedValues[key][(19 + shift).toString()].push(BulkDiscount.GetValue());
                beh.updatedValues[key][(19 + shift).toString()].push(BulkDiscount.GetValue().toString());

                beh.updatedValues[key][(20 + shift).toString()] = new Array();
                beh.updatedValues[key][(20 + shift).toString()].push(BulkDiscountTypeListBox.GetValue());
                beh.updatedValues[key][(20 + shift).toString()].push(BulkDiscountTypeListBox.GetValue().toString());

                /*
                var curr_date = d.getDate();
                    var curr_month = d.getMonth() + 1; //Months are zero based
                    var curr_year = d.getFullYear();
                */

                var onDate = BulkOnDate.GetDate();
                if (onDate != null) {
                    var day = ("0" + onDate.getDate()).slice(-2)
                    var month = ("0" + (onDate.getMonth() + 1)).slice(-2)
                    var year = onDate.getFullYear();
                    var finalOnDate = day + '/' + month + '/' + year;
                    beh.updatedValues[key][(24+shift).toString()] = new Array();
                    beh.updatedValues[key][(24+shift).toString()].push(BulkOnDate.GetDate());
                    beh.updatedValues[key][(24 + shift).toString()].push(BulkOnDate.GetDate());
                    $(row).children().eq(23 + shift).html(finalOnDate);
                    $(row).children().eq(23 + shift).css("background-color", "#D7F9C7");
                }
                var offDate = BulkOffDate.GetDate();
                if (offDate != null) {
                    var day = ("0" + offDate.getDate()).slice(-2)
                    var month = ("0" + (offDate.getMonth() + 1)).slice(-2)
                    var year = offDate.getFullYear();
                    var finalOffDate = day + '/' + month + '/' + year;
                    beh.updatedValues[key][(25+shift).toString()] = new Array();
                    beh.updatedValues[key][(25+shift).toString()].push(BulkOffDate.GetDate());
                    beh.updatedValues[key][(25+shift).toString()].push(BulkOffDate.GetDate());
                    $(row).children().eq(24 + shift).html(finalOffDate);
                    $(row).children().eq(24 + shift).css("background-color", "#D7F9C7");
                }
                if (onDate != null || offDate != null) {
                    var status = CalculateStatus(onDate, offDate);
                    $(row).children().eq(25 + shift).html(status.StatusText);
                    $(row).children().eq(25 + shift).css("background-color", "#D7F9C7");
                }

                // gross margin calculation:

                // convert COST STERLING into CURRENT CURRENCY by EX RATE. E.g. for $, *1.6
                // then it's (PRICEEX - COST in CURRENCY) / PRICEEX = 0.x  * 100, stick it in... the right box...

                var currentPriceInc = $(row).children().eq(14 + shift).html();
                var currentPriceEx = $(row).children().eq(15 + shift).html();

                var costSterling = $(row).children().eq(17 + shift).html().replace('£', '');
                var costCurrency = costSterling * ExRate;
                //$('#SalePriceGridView_DXDataRow' + (index - 1)).children().eq(14).html();

                $(row).children().eq(18 + shift).css("background-color", "#D7F9C7");
                $(row).children().eq(18 + shift).html(BulkDiscount.GetText());
                //SalePriceGridView.GetBatchEditHelper().pageServerValues[key][19] = BulkDiscount.GetValue();

                $(row).children().eq(19 + shift).css("background-color", "#D7F9C7");
                $(row).children().eq(19 + shift).html(BulkDiscountTypeListBox.GetSelectedItem().text);
                //SalePriceGridView.GetBatchEditHelper().pageServerValues[key][20] = BulkDiscountTypeListBox.GetValue();


                if (BulkDiscountTypeListBox.GetSelectedItem().value == 1) {

                    var newPriceInc = currentPriceInc * (1 - (BulkDiscount.GetValue() / 100));
                    var newPriceEx = currentPriceEx * (1 - (BulkDiscount.GetValue() / 100));

                    $(row).children().eq(20 + shift).html((Math.round(newPriceInc * 100) / 100).toFixed(2));
                    $(row).children().eq(20 + shift).css('background-color', '#D7F9C7');

                    $(row).children().eq(21 + shift).html((Math.round(newPriceEx * 100) / 100).toFixed(2));
                    $(row).children().eq(21 + shift).css('background-color', '#D7F9C7');


                    beh.updatedValues[key][(21 + shift).toString()] = new Array();
                    beh.updatedValues[key][(21 + shift).toString()].push((Math.round(newPriceInc * 100) / 100));
                    beh.updatedValues[key][(21 + shift).toString()].push((Math.round(newPriceInc * 100) / 100).toString());//.toFixed(2));

                    beh.updatedValues[key][(22 + shift).toString()] = new Array();
                    beh.updatedValues[key][(22 + shift).toString()].push((Math.round(newPriceEx * 100) / 100));
                    beh.updatedValues[key][(22 + shift).toString()].push((Math.round(newPriceEx * 100) / 100).toString());//.toFixed(2));


                    var GM = (((newPriceEx - costCurrency) / newPriceEx) * 100);
                    GM = Math.round(GM);
                    $(row).children().eq(22 + shift).html(GM + '%');
                    $(row).children().eq(22 + shift).css("background-color", "#D7F9C7");

                }
                else {

                    var newPriceInc = currentPriceInc - BulkDiscount.GetValue();
                    var newPriceEx = currentPriceEx - BulkDiscount.GetValue();

                    $(row).children().eq(20 + shift).html((Math.round(newPriceInc * 100) / 100).toFixed(2));
                    $(row).children().eq(20 + shift).css('background-color', '#D7F9C7');

                    $(row).children().eq(21 + shift).html((Math.round(newPriceEx * 100) / 100).toFixed(2));
                    $(row).children().eq(21 + shift).css('background-color', '#D7F9C7');


                    beh.updatedValues[key][(21 + shift).toString()] = new Array();
                    beh.updatedValues[key][(21 + shift).toString()].push((Math.round(newPriceInc * 100) / 100));
                    beh.updatedValues[key][(21 + shift).toString()].push((Math.round(newPriceInc * 100) / 100).toString());//.toFixed(2));

                    beh.updatedValues[key][(22 + shift).toString()] = new Array();
                    beh.updatedValues[key][(22 + shift).toString()].push((Math.round(newPriceEx * 100) / 100));
                    beh.updatedValues[key][(22 + shift).toString()].push((Math.round(newPriceEx * 100) / 100).toString());//.toFixed(2));

                    var GM = (((newPriceEx - costCurrency) / newPriceEx) * 100);

                    $(row).children().eq(22 + shift).css("background-color", "#D7F9C7");

                }
            }

                    title: "Selection",
                    width: 600,
                    height: "auto",
                    autoOpen: false,
                    resizable: false,
                    buttons: {
                        Ok: function () {
                            switch ($(this).data("type")) {
                                case "update":
                                    var keys = SalePriceGridView.GetSelectedKeysOnPage();
                                    for (var i = 0; i < keys.length; i++) {
                                        var cell = $("#SalePriceGridView tr.data-row td:first-child").filter(function () {
                                            return $(this).html() == keys[i];
                                        }).first();
                                        var index = $(cell).parent().parent().children().index($(cell).parent()) - 3;
                                        BulkUpdateValuesClientSide(keys[i], '#SalePriceGridView_DXDataRow'+index, 0);
                                    }
                                    break;

                                case "addnew":
                                    var keys = SalePriceAddNewGridView.GetSelectedKeysOnPage();

                                        var cell = $("#SalePriceAddNewGridView tr.add-new-data-row td:first-child").filter(function () {

                                    }
                                    break;

                            }
                            $(this).data("type", "");
                            
                            $(this).dialog("close");
                        },

                            $(this).dialog("close");
                        }
                    }
                });

                                    break;
                                default:
                                    $(this).dialog("close");
                                    break;
                            }
                        },

                        }
                    }

            function DeleteSelectedSalePrices() {

                var keys = null;

                if (typeof (SalePriceGridView) != "undefined" && SalePriceGridView != false)
                    keys = SalePriceGridView.GetSelectedKeysOnPage();

                if (typeof (keys) != "undefined" && keys != false) {
                    if (typeof (SalePriceGridView) != "undefined" && SalePriceGridView != false) {
                        $("#confirm-dialog").data("action", "delete");
                        $("#confirm-dialog").dialog("open");
                    }
                }
                else {
                    $("#bad-selection-dialog").dialog("open");
                }
            }

            function ConfirmedDelete() {

                SalePriceGridView.PerformCallback({ commandType: 'Delete', keyString: keys.toString() });
            }
